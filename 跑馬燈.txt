    <script>
        const panel = document.getElementById('control-panel');
        const marqueeText = document.getElementById('marquee-text');
        
        // 綁定輸入元件
        const inputs = {
            text: document.getElementById('input-text'),
            size: document.getElementById('input-size'),
            speed: document.getElementById('input-speed'),
            dir: document.getElementById('input-dir'),
            font: document.getElementById('input-font'),
            glow: document.getElementById('input-glow'),
            color: document.getElementById('input-color'),
            bg: document.getElementById('input-bg')
        };

        const displays = {
            size: document.getElementById('val-size'),
            speed: document.getElementById('val-speed')
        };

        // 更新 CSS 變數與內容的函數
        function updateDisplay() {
            marqueeText.innerText = inputs.text.value;
            
            // 更新 Root CSS 變數
            document.documentElement.style.setProperty('--font-size', `${inputs.size.value}px`);
            document.documentElement.style.setProperty('--anim-duration', `${inputs.speed.value}s`);
            document.documentElement.style.setProperty('--font-family', inputs.font.value);
            document.documentElement.style.setProperty('--text-color', inputs.color.value);
            document.documentElement.style.setProperty('--bg-color', inputs.bg.value);
            
            // 更新發光特效顏色
            const glow = inputs.glow.value;
            document.documentElement.style.setProperty('--glow-color', glow);
            
            // 更新面板的發光主題色 (讓介面也跟著變色)
            panel.style.borderColor = glow;
            panel.style.borderLeftColor = glow;
            
            // 更新方向動畫 Class
            marqueeText.className = `marquee-text ${inputs.dir.value}`;

            // 更新數值顯示
            displays.size.innerText = inputs.size.value;
            displays.speed.innerText = inputs.speed.value;
        }

        // 監聽所有輸入變化，達成即時預覽
        Object.values(inputs).forEach(input => {
            input.addEventListener('input', updateDisplay);
        });

        // 隱藏/顯示控制面板的邏輯
        function togglePanel() {
            panel.classList.toggle('hidden');
        }

        // 監聽全域 Esc 按鍵
        document.addEventListener('keydown', (e) => {
            if (e.key === 'Escape') {
                togglePanel();
                // 如果面板隱藏，讓輸入框失去焦點，避免後續誤觸
                if (panel.classList.contains('hidden')) {
                    document.activeElement.blur();
                }
            }
        });

        // 關閉按鈕事件
        document.getElementById('btn-close').addEventListener('click', togglePanel);
    </script>
